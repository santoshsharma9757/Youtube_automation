  python main.py [OPTIONS]

  OPTIONS:
    --count N          Number of short videos to generate (default: 1)
    --long             Generate a long-format video (~2.5 min)
    --image            Use image mode with ElevenLabs TTS (Required for Long)
    --upload           Enable upload to YouTube + Facebook after generation
    --skip-upload      Force skip upload
    --category CAT     Story category: magical_adventure | mythology |
                       dadi_kahani | real_life | family_funny |
                       animal_tales | mystery | seasonal | horror
    --topic "TITLE"    Force a specific story title/seed from the topic bank

  EXAMPLES:
    # Mythology long-form video
    python main.py --long --image --category mythology

    # Diwali seasonal story short
    python main.py --count 1 --image --category seasonal
