FULL COMMAND REFERENCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  python main.py [OPTIONS]

  OPTIONS:
    --count N          Number of videos to generate (default: 1)
    --videos           Use AI video generation (Veo / Google AI Studio)
    --image            Use image slideshow mode (faster, no AI video quota)
    --format FORMAT    Video format: short | mini | long | series
                       Default: short
    --upload           Enable upload to YouTube + Facebook after generation
    --skip-upload      Force skip upload (even if UPLOAD_ENABLED=true in .env)
    --dry-run          Generate story plan only (no video rendering)
    --children         Flag the video as 'Made for Kids' (monetization disabled)
    --normal           Flag the video as 'Not Made for Kids' (monetization active)
    --category CAT     Story category: magical_adventure | mythology |
                       dadi_kahani | real_life | family_funny |
                       animal_tales | mystery | seasonal | horror
    --topic "TITLE"    Force a specific story title/seed from the topic bank

  EXAMPLES:
    # Mythology long-form video (Made for Children)
    python main.py --count 1 --videos --format long --category mythology --children

    # Diwali seasonal story short (Normal/monetized)
    python main.py --count 1 --image --category seasonal --normal

    # 3 mini stories batch (normal/monetized) + upload
    python main.py --count 3 --image --format mini --normal --upload

    # Dry run to see story plan without rendering
    python main.py --count 1 --dry-run
